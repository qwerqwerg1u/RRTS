local function safeRequest(options)
    local success, res = pcall(function()
        local req = http_request or request or HttpPost or syn.request
        return req(options)
    end)
    if success then return res end
    return nil
end

local function httpGet(url)
    local success, res = pcall(function()
        return game:HttpGet(url)
    end)
    if success then return res end
    return nil
end

-- アバター画像取得

if thumbRes and thumbRes.Body then
    local decoded = HttpService:JSONDecode(thumbRes.Body)
    if decoded and decoded.data and decoded.data[1] then
        avatarUrl = decoded.data[1].imageUrl
        headshotUrl = avatarUrl:gsub("Avatar", "AvatarHeadshot")
    end
end

-- 時間計算
local function formatTime(seconds)
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    local s = math.floor(seconds % 60)
    return string.format("%d:%02d:%02d", h, m, s)
end

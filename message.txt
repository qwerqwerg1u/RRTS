function CheckBlob(blob, myHRP, myAttach, source)
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript")
    if not script then return end

    for _, side in ipairs({"Left","Right"}) do
        local detector = blob:FindFirstChild(side.."Detector")
        local weld = detector and detector:FindFirstChild(side.."Weld")
        local align = detector and detector:FindFirstChild(side.."AlignOrientation")
        local rigid = detector and detector:FindFirstChild("RigidConstraint")

        if weld and weld:IsA("AlignPosition") and weld.Attachment0 == myAttach then
            local msg = source.." → "..side.." Grab"
            local now = tick()

            if not notifyCooldowns[msg] or (now - notifyCooldowns[msg]) >= 2 then
                notifyCooldowns[msg] = now
                Rayfield:Notify({Title = "[ ✊ ]",Content = msg,Duration = 3,Image = 0})
            end

            if align and align:IsA("AlignOrientation") then
                align.Attachment0 = nil
                if weld then
                    weld.Attachment0 = nil
                end
            end

            --BlobDrop(blob, myHRP, side)
            rs.CharacterEvents.RagdollRemote:FireServer(myHRP, 0)
        end
    end
end

local function getClosestPlayer(targetPart)
    local closest, distance = nil, math.huge
    for _, player in pairs(game.Players:GetPlayers()) do
        if player ~= plr and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local mag = (targetPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
            if mag < distance then
                distance = mag
                closest = player
            end
        end
    end
    return closest
end

function AntiDeathF()
    local plr = game.Players.LocalPlayer
    local RunService = game:GetService("RunService")
    
    local connection
    connection = RunService.Heartbeat:Connect(function()
        if not AntiDeathT then
            connection:Disconnect()
            return
        end
        
        local Character = plr.Character
        if not Character then return end
        
        local inv = workspace:FindFirstChild(plr.Name.."SpawnedInToys")
        if not inv then return end
        
        local donut = inv:FindFirstChild("FoodDonut")
        if not donut or not donut:FindFirstChild("HoldPart") then return end
        
        local HoldItemRemoteFunction = donut.HoldPart:FindFirstChild("HoldItemRemoteFunction")
        local DropItemRemoteFunction = donut.HoldPart:FindFirstChild("DropItemRemoteFunction")
        
        if HoldItemRemoteFunction and DropItemRemoteFunction then
            task.spawn(function()
            pcall(function()
                HoldItemRemoteFunction:InvokeServer(donut, Character)
                DropItemRemoteFunction:InvokeServer(donut, CFrame.new(99999,99999,99999), Vector3.new(99999,99999,99999))
            end)
            end)
        end
    end)
end

local function findTargetPart()
    local grabParts = workspace:FindFirstChild("GrabParts")
    if grabParts then
        local gp = grabParts:FindFirstChild("GrabPart")
        local weld = gp and gp:FindFirstChildOfClass("WeldConstraint")
        local target = weld and weld.Part1

        if target and target:IsA("BasePart") and target.Transparency == 1 then
            target.Transparency = 0.999
        end

        return target
    end
    return nil
end

local function createHighlight(part)
    local highlight = Instance.new("Highlight")
    highlight.Name = "AutoOwner_Highlight"
    highlight.Adornee = part
    highlight.FillColor = Color3.fromRGB(0, 0, 255)
    highlight.FillTransparency = 0.75
    highlight.OutlineTransparency = 0.75
    highlight.Parent = part
    return highlight
end

local function cleanupTarget()
    if highlightInstance and highlightInstance.Parent then
        highlightInstance:Destroy()
    end
    highlightInstance = nil
    targetPart = nil
end

local AutoOwner = MTab:CreateKeybind({
    Name = "Auto-Owner",
    CurrentKeybind = "V",
    HoldToInteract = false,
    Flag = "AutoOwnerKeybind",
    Callback = function()
        AutoOwnerT = not AutoOwnerT

        if AutoOwnerT then
            targetPart = findTargetPart()

            if targetPart then
                highlightInstance = createHighlight(targetPart)
                targetPart.Destroying:Connect(cleanupTarget)
            else
                AutoOwnerT = false
            end
        else
            cleanupTarget()
        end
    end,
})

    while task.wait() do
        if AutoOwnerT and targetPart and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local myTorso = plr.Character.HumanoidRootPart

            if not targetPart.Parent then
                cleanupTarget()
                continue
            end

            if targetPart:IsA("BasePart") and targetPart.Transparency == 1 then
                targetPart.Transparency = 0.999
            end

            local dist = (targetPart.Position - myTorso.Position).Magnitude
            if dist <= 30 then
                SetNetworkOwner:FireServer(targetPart, myTorso.CFrame)
            end
        end
    end
end)
